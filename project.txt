export const colorFromString = (str = '') => {
  let hash = 0;
  for (let i = 0; i < str.length; i++) {
    hash = str.charCodeAt(i) + ((hash << 5) - hash);
  }
  return COLORS[Math.abs(hash) % COLORS.length];
};

export const initialFromString = (str = '') => {
  const s = str.trim();
  return s ? s[0].toUpperCase() : '?';
};

// Кэш профилей, чтобы не тянуть один и тот же профиль десятки раз
const cache = new Map();
const pending = new Map();

export async function getProfile(supabase, userId) {
  if (!userId) return null;
  if (cache.has(userId)) return cache.get(userId);
  if (pending.has(userId)) return pending.get(userId);

  const promise = supabase
    .from('profiles')
    .select('username, avatar_url')
    .eq('id', userId)
    .maybeSingle()
    .then(({ data }) => {
      if (data) cache.set(userId, data);
      pending.delete(userId);
      return data;
    })
    .catch(() => {
      pending.delete(userId);
      return null;
    });

  pending.set(userId, promise);
  return promise;
}

// Сбрасываем кэш, когда пользователь сохраняет новый ник/аватар
export function invalidateProfile(userId) {
  if (!userId) return;
  cache.delete(userId);
}

import { useEffect, useRef } from 'react';

/* 
 * Handles Android/browser back button.
 * Pushes a dummy history entry on mount, listens for popstate.
 * When user presses back → onBack() runs.
 * When modal unmounts programmatically, listener is removed.
 */
export function useBackButton(onBack, enabled = true) {
  const onBackRef = useRef(onBack);
  onBackRef.current = onBack;

  useEffect(() => {
    if (!enabled) return;

    try {
      window.history.pushState({ __modal: true }, '');
    } catch {
      // ignore — some browsers may block
    }

    const onPop = () => {
      onBackRef.current?.();
    };

    window.addEventListener('popstate', onPop);
    return () => {
      window.removeEventListener('popstate', onPop);
    };
  }, [enabled]);
}

import React, { useState, useEffect, useRef, useCallback } from 'react';

export default function Tabs({ tabs, active, onChange }) {
  const containerRef = useRef(null);
  const [capsule, setCapsule] = useState({ left: 0, width: 0 });
  const [dragging, setDragging] = useState(false);
  const [hoveredTab, setHoveredTab] = useState(null);

  const dragActiveRef = useRef(false);
  const skipClickRef = useRef(false);
  const stateRef = useRef({
    startX: 0,
    startLeft: 0,
    currentLeft: 0,
    moved: false,
    positions: [],
    capsuleWidth: 0,
    active,
    onChange,
  });

  stateRef.current.active = active;
  stateRef.current.onChange = onChange;
  stateRef.current.capsuleWidth = capsule.width;

  const collectPositions = useCallback(() => {
    const container = containerRef.current;
    if (!container) return [];
    return Array.from(container.querySelectorAll('.tab')).map((btn) => ({
      id: btn.dataset.tabId,
      left: btn.offsetLeft,
      width: btn.offsetWidth,
      center: btn.offsetLeft + btn.offsetWidth / 2,
    }));
  }, []);

  const snapCapsuleTo = useCallback((positions, tabId) => {
    const btn = positions.find((p) => p.id === tabId);
    if (btn) setCapsule({ left: btn.left, width: btn.width });
  }, []);

  useEffect(() => {
    const positions = collectPositions();
    stateRef.current.positions = positions;
    snapCapsuleTo(positions, active);
  }, [active, tabs, collectPositions, snapCapsuleTo]);

  useEffect(() => {
    const onResize = () => {
      const positions = collectPositions();
      stateRef.current.positions = positions;
      snapCapsuleTo(positions, stateRef.current.active);
    };
    window.addEventListener('resize', onResize);
    return () => window.removeEventListener('resize', onResize);
  }, [collectPositions, snapCapsuleTo]);

  const handlePointerDown = (e) => {
    if (e.pointerType === 'mouse' && e.button !== 0) return;

    const positions = collectPositions();
    stateRef.current.positions = positions;
    stateRef.current.startX = e.clientX;
    stateRef.current.startLeft = capsule.left;
    stateRef.current.currentLeft = capsule.left;
    stateRef.current.moved = false;
    dragActiveRef.current = true;
    setDragging(true);
  };

  useEffect(() => {
    if (!dragging) return;

    const onMove = (e) => {
      if (!dragActiveRef.current) return;
      const dx = e.clientX - stateRef.current.startX;
      if (Math.abs(dx) > 5) stateRef.current.moved = true;

      const positions = stateRef.current.positions;
      if (positions.length === 0) return;

      let newLeft = stateRef.current.startLeft + dx;
      const minLeft = positions[0].left;
      const maxLeft = positions[positions.length - 1].left;
      if (newLeft < minLeft) newLeft = minLeft;
      if (newLeft > maxLeft) newLeft = maxLeft;

      stateRef.current.currentLeft = newLeft;
      setCapsule((prev) => ({ ...prev, left: newLeft }));

      const center = newLeft + stateRef.current.capsuleWidth / 2;
      let nearest = positions[0];
      let minDist = Math.abs(center - nearest.center);
      for (const p of positions) {
        const dist = Math.abs(center - p.center);
        if (dist < minDist) {
          minDist = dist;
          nearest = p;
        }
      }
      setHoveredTab(nearest.id);
    };

    const onUp = () => {
      dragActiveRef.current = false;
      setDragging(false);

      const positions = stateRef.current.positions;

      if (!stateRef.current.moved) {
        setHoveredTab(null);
        if (positions.length) snapCapsuleTo(positions, stateRef.current.active);
        return;
      }

      skipClickRef.current = true;
      setTimeout(() => { skipClickRef.current = false; }, 80);

      if (positions.length === 0) {
        setHoveredTab(null);
        return;
      }

      const center = stateRef.current.currentLeft + stateRef.current.capsuleWidth / 2;
      let nearest = positions[0];
      let minDist = Math.abs(center - nearest.center);
      for (const p of positions) {
        const dist = Math.abs(center - p.center);
        if (dist < minDist) {
          minDist = dist;
          nearest = p;
        }
      }

      setHoveredTab(null);
      snapCapsuleTo(positions, nearest.id);

      if (nearest.id !== stateRef.current.active) {
        stateRef.current.onChange(nearest.id);
      }
    };

    window.addEventListener('pointermove', onMove);
    window.addEventListener('pointerup', onUp);
    window.addEventListener('pointercancel', onUp);
    return () => {
      window.removeEventListener('pointermove', onMove);
      window.removeEventListener('pointerup', onUp);
      window.removeEventListener('pointercancel', onUp);
    };
  }, [dragging, snapCapsuleTo]);

  const handleClick = (tabId) => {
    if (skipClickRef.current) return;
    if (dragActiveRef.current) return;
    onChange(tabId);
  };

  return (
    <div className="tabs" ref={containerRef} onPointerDown={handlePointerDown}>
      <div
        className={`tabs-capsule ${dragging ? 'is-dragging' : ''}`}
        style={{
          left: `${capsule.left}px`,
          width: `${capsule.width}px`,
          transform: `scale(${dragging ? 1.08 : 1})`,
        }}
      />
      {tabs.map((t) => {
        const isVisualActive = dragging ? hoveredTab === t.id : active === t.id;
        return (
          <button
            key={t.id}
            data-tab-id={t.id}
            className={`tab ${isVisualActive ? 'is-active' : ''}`}
            onClick={() => handleClick(t.id)}
          >
            {t.label}
          </button>
        );
      })}
    </div>
  );
}

import React, { useState, useEffect, useRef, useCallback } from 'react';
import './AudioPlayer.css';

const formatTime = (sec) => {
  if (!sec || !isFinite(sec)) return '0:00';
  const m = Math.floor(sec / 60);
  const s = Math.floor(sec % 60).toString().padStart(2, '0');
  return `${m}:${s}`;
};

export default function AudioPlayer({ src, name }) {
  const audioRef = useRef(null);
  const waveformRef = useRef(null);

  const [playing, setPlaying] = useState(false);
  const [current, setCurrent] = useState(0);
  const [duration, setDuration] = useState(0);
  const [peaks, setPeaks] = useState([]);
  const [scrubbing, setScrubbing] = useState(false);

  const scrubRef = useRef({ active: false, pointerId: null });

  /* decode audio and extract peaks for waveform */
  useEffect(() => {
    let cancelled = false;
    let ctx = null;

    const load = async () => {
      try {
        const res = await fetch(src);
        const arrayBuffer = await res.arrayBuffer();
        ctx = new (window.AudioContext || window.webkitAudioContext)();
        const audioBuffer = await ctx.decodeAudioData(arrayBuffer);

        if (cancelled) return;

        const channel = audioBuffer.getChannelData(0);
        const blockSize = Math.floor(channel.length / 40);
        const extracted = [];

        for (let i = 0; i < 40; i++) {
          let sum = 0;
          const start = i * blockSize;
          for (let j = 0; j < blockSize; j++) {
            sum += Math.abs(channel[start + j] || 0);
          }
          extracted.push(sum / blockSize);
        }

        const max = Math.max(...extracted, 0.001);
        setPeaks(extracted.map((p) => p / max));
      } catch {
        /* fallback: static peaks */
        setPeaks(Array.from({ length: 40 }, () => 0.2 + Math.random() * 0.8));
      } finally {
        if (ctx) ctx.close();
      }
    };

    load();
    return () => { cancelled = true; };
  }, [src]);

  /* time tracking */
  useEffect(() => {
    const audio = audioRef.current;
    if (!audio) return;

    const onTime = () => {
      if (!scrubRef.current.active) setCurrent(audio.currentTime);
    };
    const onMeta = () => setDuration(audio.duration);
    const onEnd = () => {
      setPlaying(false);
      setCurrent(0);
    };

    audio.addEventListener('timeupdate', onTime);
    audio.addEventListener('loadedmetadata', onMeta);
    audio.addEventListener('ended', onEnd);
    return () => {
      audio.removeEventListener('timeupdate', onTime);
      audio.removeEventListener('loadedmetadata', onMeta);
      audio.removeEventListener('ended', onEnd);
    };
  }, []);

  const toggle = useCallback(() => {
    const audio = audioRef.current;
    if (!audio) return;
    if (playing) {
      audio.pause();
      setPlaying(false);
    } else {
      audio.play().catch(() => {});
      setPlaying(true);
    }
  }, [playing]);

  /* ---------- scrubbing ---------- */
  const seekToX = useCallback((clientX) => {
    const el = waveformRef.current;
    const audio = audioRef.current;
    if (!el || !audio || !duration) return;

    const rect = el.getBoundingClientRect();
    const x = Math.max(0, Math.min(rect.width, clientX - rect.left));
    const ratio = x / rect.width;
    const newTime = ratio * duration;

    audio.currentTime = newTime;
    setCurrent(newTime);
  }, [duration]);

  const handleScrubDown = (e) => {
    if (e.pointerType === 'mouse' && e.button !== 0) return;
    scrubRef.current.active = true;
    scrubRef.current.pointerId = e.pointerId;
    setScrubbing(true);
    try { waveformRef.current?.setPointerCapture(e.pointerId); } catch {}
    seekToX(e.clientX);
  };

  const handleScrubMove = (e) => {
    if (!scrubRef.current.active) return;
    if (e.pointerId !== scrubRef.current.pointerId) return;
    seekToX(e.clientX);
  };

  const handleScrubUp = (e) => {
    if (!scrubRef.current.active) return;
    if (e.pointerId !== scrubRef.current.pointerId) return;
    try { waveformRef.current?.releasePointerCapture(e.pointerId); } catch {}
    scrubRef.current.active = false;
    setScrubbing(false);
  };

  const progress = duration ? current / duration : 0;
  const activeBars = Math.floor(progress * peaks.length);

  /* single time label: total when idle, elapsed when playing or scrubbing */
  const showElapsed = playing || scrubbing || current > 0;
  const timeLabel = showElapsed
    ? `${formatTime(current)} / ${formatTime(duration)}`
    : formatTime(duration);

  return (
    <div className="audio-player">
      <button className="audio-play" onClick={toggle} aria-label={playing ? 'Пауза' : 'Воспроизвести'}>
        {playing ? (

import React, { useState, useEffect, useRef } from 'react';

import Avatar from '../../Feed/components/post/Avatar';
import { TrashIcon, VerifiedIcon } from '../../../assets/icons/Icons';

const VERIFIED_EMAILS = ['obrekos6@gmail.com'];

export default function CommentItem({ comment, currentUser, onDelete }) {
  const [author, setAuthor] = useState(null);
  const mountedRef = useRef(true);

  const isAuthor = !!(currentUser && comment.user_id === currentUser.id);
  const isTemp = String(comment.id).startsWith('temp-');
  const isVerified = !!(comment.author_email && VERIFIED_EMAILS.includes(comment.author_email));

  useEffect(() => {
    mountedRef.current = true;
    return () => { mountedRef.current = false; };
  }, []);

  useEffect(() => {
    if (!comment.user_id) return;
    let cancelled = false;
    getProfile(supabase, comment.user_id).then((data) => {
      if (!cancelled && mountedRef.current && data) setAuthor(data);
    });
    return () => { cancelled = true; };
  }, [comment.user_id]);

  const fallback = comment.user_id ? comment.user_id.slice(0, 6) : 'user';
  const displayName = author?.username || fallback;

  const handleDelete = () => {
    if (!window.confirm('Удалить комментарий?')) return;
    onDelete(comment.id);
  };

  return (
    <div className={`comment-item ${isTemp ? 'is-temp' : ''}`}>
      <div className="comment-item-avatar">
        <Avatar
          url={author?.avatar_url}
          name={displayName}
          seed={comment.user_id}
          size={36}
        />
      </div>
      <div className="comment-body">
        <div className="comment-head">
          <span className="comment-author">{displayName}</span>
          {isVerified && <VerifiedIcon size={14} className="comment-verified" />}
          <span className="comment-time">{formatTime(comment.created_at)}</span>
        </div>
        <div className="comment-text">{comment.content}</div>
      </div>
      {isAuthor && !isTemp && (
        <button className="comment-delete" onClick={handleDelete} aria-label="Удалить">
          <TrashIcon size={16} />
        </button>
      )}
    </div>
  );
}

import React, { useEffect, useState, useRef, useCallback, useLayoutEffect } from 'react';
import { createPortal } from 'react-dom';
import { supabase } from '../../lib/supabaseClient';
import { useBackButton } from '../../lib/useBackButton';
import CommentList from './components/CommentList';
import CommentInput from './components/CommentInput';
import './PostDetailScreen.css';

export default function PostDetailScreen({ post, currentUser, onClose }) {
  const [comments, setComments] = useState([]);
  const [loading, setLoading] = useState(true);
  const [visible, setVisible] = useState(false);
  const [closing, setClosing] = useState(false);

  const [dragging, setDragging] = useState(false);
  const [dragOffset, setDragOffset] = useState(0);

  const dragRef = useRef({
    startY: 0,
    currentY: 0,
    active: false,
    pointerId: null,
    lastY: 0,
    lastT: 0,
    velocity: 0,
  });
  const panelRef = useRef(null);
  const handleZoneRef = useRef(null);

  const mediaFiles = (post.files || []).filter(
    (f) => f.type?.startsWith('image/') || f.type?.startsWith('video/')
  );
  const hasMedia = mediaFiles.length > 0;
  const firstMedia = mediaFiles[0];

  /* load comments */
  const load = useCallback(async () => {
    setLoading(true);
    const { data, error } = await supabase
      .from('comments')
      .select('*')
      .eq('repository_id', post.id)
      .order('created_at', { ascending: false });
    if (!error) setComments(data || []);
    setLoading(false);
  }, [post.id]);

  useEffect(() => { load(); }, [load]);

  /* realtime */
  useEffect(() => {
    const ch = supabase
      .channel(`comments-${post.id}`)
      .on('postgres_changes',
        { event: 'INSERT', schema: 'public', table: 'comments', filter: `repository_id=eq.${post.id}` },
        (p) => setComments((prev) => {
          const tempIdx = prev.findIndex(
            (c) => String(c.id).startsWith('temp-') && c.user_id === p.new.user_id
          );
          if (tempIdx >= 0) {
            const copy = [...prev];
            copy[tempIdx] = p.new;
            return copy;
          }
          if (prev.some((c) => c.id === p.new.id)) return prev;
          return [p.new, ...prev];
        }))
      .on('postgres_changes',
        { event: 'DELETE', schema: 'public', table: 'comments', filter: `repository_id=eq.${post.id}` },
        (p) => setComments((prev) => prev.filter((c) => c.id !== p.old.id)))
      .subscribe();
    return () => supabase.removeChannel(ch);
  }, [post.id]);

  /* lock body scroll */
  useEffect(() => {
    const prev = document.body.style.overflow;
    document.body.style.overflow = 'hidden';
    return () => { document.body.style.overflow = prev; };
  }, []);

  /* esc */
  useEffect(() => {
    const onKey = (e) => { if (e.key === 'Escape') handleClose(); };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, []);

  /* slide up */
  useLayoutEffect(() => {
    requestAnimationFrame(() => requestAnimationFrame(() => setVisible(true)));
  }, []);

  const handleClose = () => {
    if (closing) return;
    setClosing(true);
    setVisible(false);
    setDragging(false);
    setDragOffset(0);
    setTimeout(() => onClose(), 320);
  };

  /* android back button closes */
  useBackButton(handleClose, true);

  /* ---------- drag to close ---------- */
  const handleDragStart = (e) => {
    if (e.pointerType === 'mouse' && e.button !== 0) return;
    dragRef.current.active = true;
    dragRef.current.startY = e.clientY;
    dragRef.current.currentY = e.clientY;
    dragRef.current.lastY = e.clientY;
    dragRef.current.lastT = performance.now();
    dragRef.current.velocity = 0;
    dragRef.current.pointerId = e.pointerId;

    setDragging(true);
    try { handleZoneRef.current?.setPointerCapture(e.pointerId); } catch {}
  };

  const handleDragMove = (e) => {
    if (!dragRef.current.active) return;
    if (e.pointerId !== dragRef.current.pointerId) return;

    const now = performance.now();
    const dt = now - dragRef.current.lastT;
    const dy = e.clientY - dragRef.current.lastY;
    if (dt > 0) {
      dragRef.current.velocity = dy / dt;
    }
    dragRef.current.lastY = e.clientY;
    dragRef.current.lastT = now;
    dragRef.current.currentY = e.clientY;

    const totalDy = e.clientY - dragRef.current.startY;
    if (totalDy < 0) {
      setDragOffset(0);
      return;
    }
    setDragOffset(totalDy);
  };

  const handleDragEnd = (e) => {
    if (!dragRef.current.active) return;
    if (e.pointerId !== dragRef.current.pointerId) return;

    try { handleZoneRef.current?.releasePointerCapture(e.pointerId); } catch {}
    dragRef.current.active = false;
    setDragging(false);

    const totalDy = dragRef.current.currentY - dragRef.current.startY;
    const v = dragRef.current.velocity;

    if (totalDy > 60 || v > 0.4) {
      handleClose();
    } else {
      setDragOffset(0);
    }
  };

  /* ---------- send ---------- */
  const handleSend = async (text) => {
    if (!currentUser || !text.trim()) return;
    const tempId = `temp-${Date.now()}`;
    const optimistic = {
      id: tempId,
      repository_id: post.id,
      user_id: currentUser.id,
      author_email: currentUser.email,
      content: text.trim(),
      created_at: new Date().toISOString(),
    };
    setComments((prev) => [optimistic, ...prev]);

    const { data, error } = await supabase
      .from('comments')
      .insert({
        repository_id: post.id,
        user_id: currentUser.id,
        author_email: currentUser.email,
        content: text.trim(),
      })
      .select()
      .single();

    if (error) {
      setComments((prev) => prev.filter((c) => c.id !== tempId));
      alert('Не удалось отправить: ' + error.message);
      return;
    }

    setComments((prev) => prev.map((c) => (c.id === tempId ? data : c)));
  };

  const handleDeleteComment = async (id) => {
    setComments((prev) => prev.filter((c) => c.id !== id));
    const { error } = await supabase.from('comments').delete().eq('id', id);
    if (error) {
      load();
      alert('Не удалось удалить: ' + error.message);
    }
  };

  const panelStyle = dragging
    ? { transform: `translateY(${dragOffset}px)`, transition: 'none' }
    : undefined;

  return createPortal(
    <div className={`post-detail-root ${visible ? 'is-on' : ''}`}>
      <div className="post-detail-backdrop" onClick={handleClose} />

      <div
        ref={panelRef}
        className={`post-detail-panel ${hasMedia ? '' : 'is-solo'}`}
        style={panelStyle}
      >
        {hasMedia && firstMedia && (
          <div className="post-detail-media">
            {firstMedia.type.startsWith('video/') ? (
              <video src={firstMedia.url} autoPlay muted loop playsInline />
            ) : (
              <img src={firstMedia.url} alt={post.name || ''} draggable={false} />
            )}
          </div>
        )}

        <div className="post-detail-sheet">
          <div
            ref={handleZoneRef}
            className="post-detail-handle-zone"
            onPointerDown={handleDragStart}
            onPointerMove={handleDragMove}
            onPointerUp={handleDragEnd}
            onPointerCancel={handleDragEnd}
          >
            <div className="post-detail-handle" />
          </div>

          <div className="post-detail-scroll">
            {loading ? (
              <div className="post-detail-empty">Загрузка…</div>
            ) : comments.length === 0 ? (
              <div className="post-detail-empty">Пока нет комментариев</div>
            ) : (
              <CommentList
                comments={comments}
                currentUser={currentUser}
                onDelete={handleDeleteComment}
              />
            )}
          </div>

          <CommentInput onSend={handleSend} disabled={!currentUser} user={currentUser} />

import React, { useState, useEffect, useRef } from 'react';
import { supabase } from '../../../lib/supabaseClient';
import { invalidateProfile } from '../../../lib/profilesCache';
import Avatar from '../../Feed/components/post/Avatar';
import { CameraIcon } from '../../../assets/icons/Icons';
import './ProfileSetup.css';

export default function ProfileSetup({ onComplete, submitLabel = 'Продолжить' }) {
  const [user, setUser] = useState(null);
  const [avatarUrl, setAvatarUrl] = useState(null);
  const [username, setUsername] = useState('');
  const [displayName, setDisplayName] = useState('');
  const [uploading, setUploading] = useState(false);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState('');
  const fileRef = useRef(null);

  useEffect(() => {
    let cancelled = false;

    supabase.auth.getUser().then(({ data: { user } }) => {
      if (cancelled || !user) return;
      setUser(user);

      const fallback = user.email ? user.email.split('@')[0] : 'user';

      supabase
        .from('profiles')
        .select('*')
        .eq('id', user.id)
        .maybeSingle()
        .then(({ data }) => {
          if (cancelled) return;
          if (data) {
            setAvatarUrl(data.avatar_url || null);
            setUsername(data.username || fallback);
            setDisplayName(data.display_name || data.username || fallback);
          } else {
            setUsername(fallback);
            setDisplayName(fallback);
          }
        });
    });

    return () => { cancelled = true; };
  }, []);

  const handleAvatar = async (e) => {
    const file = e.target.files?.[0];
    if (!file || !user) return;
    if (file.size > 5 * 1024 * 1024) {
      alert('Файл слишком большой (макс. 5 МБ)');
      return;
    }
    setUploading(true);

    try {
      const ext = file.name.split('.').pop();
      const path = `${user.id}/${Date.now()}.${ext}`;

      const { error: upErr } = await supabase.storage
        .from('avatars')
        .upload(path, file, { cacheControl: '31536000', upsert: true });
      if (upErr) throw upErr;

      const { data: { publicUrl } } = supabase.storage
        .from('avatars')
        .getPublicUrl(path);

      setAvatarUrl(publicUrl);
      invalidateProfile(user.id);
    } catch (err) {
      alert('Ошибка: ' + err.message);
    } finally {
      setUploading(false);
    }
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError('');

    if (!user) return;

    const u = username.trim().toLowerCase().replace(/[^a-z0-9_]/g, '');
    const d = displayName.trim();

    if (u.length < 3) {
      setError('Юзернейм: минимум 3 символа (a-z, 0-9, _)');
      return;
    }
    if (!d) {
      setError('Введите имя');
      return;
    }

    setSaving(true);

    /* uniqueness check */
    const { data: existing } = await supabase
      .from('profiles')
      .select('id')
      .ilike('username', u)
      .neq('id', user.id)
      .maybeSingle();

    if (existing) {
      setError('Этот юзернейм уже занят');
      setSaving(false);
      return;
    }

    const { error: saveErr } = await supabase
      .from('profiles')
      .upsert({
        id: user.id,
        username: u,
        display_name: d,
        avatar_url: avatarUrl,
        email: user.email,
        updated_at: new Date().toISOString(),
      });

    setSaving(false);

    if (saveErr) {
      setError(saveErr.message);
      return;
    }

    invalidateProfile(user.id);
    onComplete?.();
  };

  const previewName = displayName || username || 'user';
  const previewUsername = username || '';
  const isSubmitDisabled = saving || !displayName.trim() || username.trim().length < 3;

  return (
    <form className="profile-setup" onSubmit={handleSubmit}>
      <div className="profile-setup-preview">
        <div className="profile-setup-avatar">
          {avatarUrl ? (
            <img src={avatarUrl} alt="" className="profile-setup-avatar-img" />
          ) : (
            <Avatar url={null} name={previewName} seed={user?.id} size={96} />
          )}
          <button
            type="button"
            className="profile-setup-avatar-btn"
            onClick={() => fileRef.current?.click()}
            disabled={uploading}
            aria-label="Сменить аватар"
          >
            {uploading ? '…' : <CameraIcon size={16} />}
          </button>
          <input
            ref={fileRef}
            type="file"
            accept="image/*"
            onChange={handleAvatar}
            style={{ display: 'none' }}
          />
        </div>
        <div className="profile-setup-preview-name">{previewName}</div>
        {previewUsername && (
          <div className="profile-setup-preview-username">@{previewUsername}</div>
        )}
      </div>

      <label className="profile-setup-field">
        <span className="profile-setup-label">Имя</span>
        <input
          type="text"
          className="profile-setup-input"
          value={displayName}
          onChange={(e) => setDisplayName(e.target.value)}
          placeholder="Как вас зовут"
          maxLength={40}
          required
        />
      </label>

      <label className="profile-setup-field">
        <span className="profile-setup-label">Юзернейм</span>
        <div className="profile-setup-username-wrap">
          <span className="profile-setup-at">@</span>
          <input
            type="text"
            className="profile-setup-input profile-setup-input-username"
            value={username}
            onChange={(e) =>
              setUsername(e.target.value.toLowerCase().replace(/[^a-z0-9_]/g, ''))
            }
            placeholder="username"
            maxLength={30}
            required
          />
        </div>
      </label>

      {error && <div className="profile-setup-error">{error}</div>}

      <button
        type="submit"
        className="profile-setup-submit"
        disabled={isSubmitDisabled}
      >
        {saving ? '…' : submitLabel}
      </button>
    </form>
  );
}

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
};

serve(async (req) => {
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }

  try {
    const { password } = await req.json();

    if (!password) {
      return new Response(JSON.stringify({ success: false, error: "Пароль не предоставлен" }), {
        status: 400,
        headers: { ...corsHeaders, "Content-Type": "application/json" },
      });
    }

    // Читаем секрет из переменных окружения Edge Function
    const masterPassword = Deno.env.get("REPOPART_MASTER_PASSWORD");
    
    console.log("Секрет найден:", masterPassword ? "да" : "нет");
    console.log("Пароль совпадает:", password === masterPassword);

    const isValid = password === masterPassword;

    return new Response(JSON.stringify({ success: isValid }), {
      headers: { ...corsHeaders, "Content-Type": "application/json" },
    });

  } catch (err) {
    console.error("Критическая ошибка:", err);
    return new Response(JSON.stringify({ success: false, error: "Внутренняя ошибка" }), {
      status: 500,
      headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  }
});

export function formatTime(dateString) {
  const diff = Math.floor((Date.now() - new Date(dateString)) / 1000);

  if (diff < 5) return 'только что';
  if (diff < 60) return `${diff} секунд назад`;

  const plural = (n, forms) => {
    const d = n % 10, dd = n % 100;
    if (dd >= 11 && dd <= 14) return forms[2];
    if (d === 1) return forms[0];
    if (d >= 2 && d <= 4) return forms[1];
    return forms[2];
  };

  const m = Math.floor(diff / 60);
  if (m < 60) return `${m} ${plural(m, ['минуту', 'минуты', 'минут'])} назад`;

  const h = Math.floor(m / 60);
  if (h < 24) return `${h} ${plural(h, ['час', 'часа', 'часов'])} назад`;

  const d = Math.floor(h / 24);
  if (d < 30) return `${d} ${plural(d, ['день', 'дня', 'дней'])} назад`;

  const mo = Math.floor(d / 30);
  if (mo < 12) return `${mo} ${plural(mo, ['месяц', 'месяца', 'месяцев'])} назад`;

  const y = Math.floor(mo / 12);
  return `${y} ${plural(y, ['год', 'года', 'лет'])} назад`;
}

import React, { useId } from 'react';
import { colorFromString, initialFromString } from '../../../../lib/avatar';
import './Avatar.css';

export default function Avatar({ url, name, seed, size = 40, className = '' }) {
  const uid = useId().replace(/:/g, '');
  const clipId = `clip-${uid}`;
  const gradId = `grad-${uid}`;

  if (url) {
    return (
      <img
        src={url}
        alt={name}
        className={`avatar ${className}`}
        style={{ width: size, height: size }}
      />
    );
  }

  // seed (id пользователя) — стабильный идентификатор. Если его нет — используем name.
  // Так цвет не меняется при смене ника.
  const colorSeed = seed || name || '';
  const color = colorFromString(colorSeed);
  const letter = initialFromString(name);

  return (
    <svg
      width={size}
      height={size}
      viewBox="0 0 104 104"
      className={`avatar avatar-fallback ${className}`}
      aria-label={name}
    >
      <defs>
        <linearGradient
          id={gradId}
          x1="52" y1="0" x2="52" y2="104"
          gradientUnits="userSpaceOnUse"
        >
          <stop stopColor="white" />
          <stop offset="1" stopColor="white" stopOpacity="0" />
        </linearGradient>
        <clipPath id={clipId}>
          <rect width="104" height="104" rx="52" fill="white" />
        </clipPath>
      </defs>
      <g clipPath={`url(#${clipId})`}>
        <rect width="104" height="104" rx="52" fill={color} />
        <g style={{ mixBlendMode: 'screen' }}>
          <rect
            width="104"
            height="104"
            fill={`url(#${gradId})`}
            fillOpacity="0.56"
          />
        </g>
      </g>
      <text
        x="52"
        y="52"
        textAnchor="middle"
        dominantBaseline="central"
        fontSize="45"
        fill="#ffffff"
        fontFamily="var(--font-nickname)"
      >
        {letter}
      </text>
    </svg>
  );
}

import React, { useEffect, useRef, useState } from 'react';
import { createPortal } from 'react-dom';
import { ShareIcon, CopyIcon, TrashIcon } from '../../../../assets/icons/Icons';
import './PostContextMenu.css';

export default function PostContextMenu({
  open,
  onClose,
  onShare,
  onCopy,
  onDelete,
  isAuthor,
  hasCaption,
}) {
  const [visible, setVisible] = useState(false);
  const sheetRef = useRef(null);
  const dragRef = useRef({
    startY: 0,
    currentY: 0,
    active: false,
    pointerId: null,
  });
  const [dragOffset, setDragOffset] = useState(0);

  useEffect(() => {
    if (open) {
      requestAnimationFrame(() => requestAnimationFrame(() => setVisible(true)));
      document.body.style.overflow = 'hidden';
    } else {
      setVisible(false);
      setDragOffset(0);
      document.body.style.overflow = '';
    }
    return () => { document.body.style.overflow = ''; };
  }, [open]);

  useEffect(() => {
    if (!open) return;
    const onKey = (e) => { if (e.key === 'Escape') handleClose(); };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [open]);

  const handleClose = () => {
    setVisible(false);
    setTimeout(() => {
      setDragOffset(0);
      onClose();
    }, 280);
  };

  const handlePointerDown = (e) => {
    if (e.pointerType === 'mouse' && e.button !== 0) return;
    const target = e.target.closest('.post-ctx-item');
    if (target) return;

    dragRef.current.active = true;
    dragRef.current.startY = e.clientY;
    dragRef.current.currentY = e.clientY;
    dragRef.current.pointerId = e.pointerId;

    try { sheetRef.current?.setPointerCapture(e.pointerId); } catch {}
  };

  const handlePointerMove = (e) => {
    if (!dragRef.current.active) return;
    if (e.pointerId !== dragRef.current.pointerId) return;
    const dy = e.clientY - dragRef.current.startY;
    if (dy < 0) return;
    dragRef.current.currentY = e.clientY;
    setDragOffset(dy);
  };

  const handlePointerUp = (e) => {
    if (!dragRef.current.active) return;
    if (e.pointerId !== dragRef.current.pointerId) return;

    try { sheetRef.current?.releasePointerCapture(e.pointerId); } catch {}

    dragRef.current.active = false;
    const dy = dragRef.current.currentY - dragRef.current.startY;

    if (dy > 120) {
      handleClose();
    } else {
      setDragOffset(0);
    }
  };

  if (!open) return null;

  const translateY = visible ? dragOffset : 400;

  return createPortal(
    <div
      className={`post-ctx-backdrop ${visible ? 'is-visible' : ''}`}
      onClick={handleClose}
    >
      <div
        ref={sheetRef}
        className={`post-ctx-sheet ${visible ? 'is-visible' : ''} ${dragRef.current.active ? 'is-dragging' : ''}`}
        style={{
          transform: `translate(-50%, ${translateY}px)`,
        }}
        onClick={(e) => e.stopPropagation()}
        onPointerDown={handlePointerDown}
        onPointerMove={handlePointerMove}
        onPointerUp={handlePointerUp}
        onPointerCancel={handlePointerUp}
      >
        <div className="post-ctx-handle" />

        <div className="post-ctx-list">
          <button
            className="post-ctx-item"
            onClick={() => { handleClose(); onShare?.(); }}
          >
            <span className="post-ctx-icon"><ShareIcon size={22} /></span>
            Поделиться
          </button>

          {hasCaption && (
            <button
              className="post-ctx-item"
              onClick={() => { handleClose(); onCopy?.(); }}
            >
              <span className="post-ctx-icon"><CopyIcon size={22} /></span>
              Скопировать
            </button>
          )}

          {isAuthor && (
            <button
              className="post-ctx-item is-danger"
              onClick={() => { handleClose(); onDelete?.(); }}
            >
              <span className="post-ctx-icon"><TrashIcon size={22} /></span>
              Удалить
            </button>
          )}
        </div>
      </div>
    </div>,
    document.body
  );
}

import { supabase } from '../../lib/supabaseClient';
import Logo from '../../assets/icons/Logo';
import { UserIcon, PlusIcon } from '../../assets/icons/Icons';
import PublicationList from '../Feed/components/post/PublicationList';
import SideMenu from '../Feed/components/layout/SideMenu';
import Tabs from '../Feed/components/layout/Tabs';
import './LandingScreen.css';

const POSTS_PER_PAGE = 10;

const NAV_ITEMS = [
  { label: 'Лента', icon: '📡', action: 'feed' },
  { label: 'Профиль', icon: '👤', action: 'profile' },
  { label: 'Мои посты', icon: '📁', action: 'my' },
  { label: 'Настройки', icon: '⚙️', action: 'settings' },
];

const TABS = [
  { id: 'for-you', label: 'Для вас' },
  { id: 'fresh', label: 'Свежее' },
  { id: 'top', label: 'Топ' },
];

export default function LandingScreen({ user, loading }) {
  const navigate = useNavigate();
  const [posts, setPosts] = useState([]);
  const [postsLoading, setPostsLoading] = useState(true);
  const [hasMore, setHasMore] = useState(true);
  const [tab, setTab] = useState('for-you');
  const [menuOpen, setMenuOpen] = useState(false);

  const sentinelRef = useRef(null);
  const loadingRef = useRef(false);

  const authed = !loading && user;

  const goAuth = useCallback(() => navigate('/auth'), [navigate]);
  const goFeed = useCallback(() => navigate('/feed'), [navigate]);
  const guestClick = useCallback(
    () => (authed ? goFeed() : goAuth()),
    [authed, goAuth, goFeed]
  );

  const loadPosts = useCallback(async (mode, limit, offset) => {
    if (mode === 'top') {
      const { data } = await supabase
        .from('repositories')
        .select('*')
        .order('hot_score', { ascending: false, nullsFirst: false })
        .order('created_at', { ascending: false })
        .range(offset, offset + limit - 1);
      return data || [];
    }

    /* both for-you and fresh — most recent first */
    const { data } = await supabase
      .from('repositories')
      .select('*')
      .order('created_at', { ascending: false })
      .range(offset, offset + limit - 1);
    return data || [];
  }, []);

  const load = useCallback(async (reset = false) => {
    if (!reset && !hasMore) return;
    if (loadingRef.current) return;

    loadingRef.current = true;
    if (reset) {
      setPosts([]);
      setPostsLoading(true);
      setHasMore(true);
    }

    const offset = reset ? 0 : posts.length;
    const data = await loadPosts(tab, POSTS_PER_PAGE, offset);

    if (reset) setPosts(data);
    else setPosts((prev) => [...prev, ...data]);

    setHasMore(data.length === POSTS_PER_PAGE);
    setPostsLoading(false);
    loadingRef.current = false;
  }, [tab, posts.length, hasMore, loadPosts]);

  useEffect(() => {
    load(true);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [tab]);

  useEffect(() => {
    if (!sentinelRef.current || !hasMore) return;
    const observer = new IntersectionObserver(
      (entries) => {
        if (entries[0].isIntersecting) load(false);
      },
      { rootMargin: '400px' }
    );
    observer.observe(sentinelRef.current);
    return () => observer.disconnect();
  }, [hasMore, load]);

  const handleLogoClick = useCallback(() => {
    if (window.scrollY > 5) {
      window.scrollTo({ top: 0, behavior: 'smooth' });
      return;
    }
    load(true);
  }, [load]);

  const handleNavClick = useCallback((action) => {
    setMenuOpen(false);
    if (action === 'feed') {
      window.scrollTo({ top: 0, behavior: 'smooth' });
      return;
    }
    guestClick();
  }, [guestClick]);

  return (
    <>
      <div className="app">
        <div className="app-bg" />

        <aside className="desktop-sidebar">
          <ul className="side-menu-list">
            {NAV_ITEMS.map((item) => (
              <li key={item.label}>
                <button
                  className="side-menu-item"
                  onClick={() => handleNavClick(item.action)}
                >
                  <span className="side-menu-icon">{item.icon}</span>
                  {item.label}
                </button>
              </li>
            ))}
          </ul>
        </aside>

        <button className="menu-btn" aria-label="Меню" onClick={() => setMenuOpen(true)}>
          <div className="menu-btn-stripes">
            <span></span><span></span><span></span>
          </div>
        </button>

        <button className="fixed-logo" onClick={handleLogoClick} aria-label="связь">
          <Logo className="logo-icon" />
          <span className="logo-text">связь</span>
          <span className="logo-beta-capsule">BETA</span>
        </button>

        <div className="double-btn">
          <button
            className="double-btn-item"
            aria-label="Войти"
            title="Войти"
            onClick={guestClick}
          >
            <UserIcon size={20} />
          </button>
          <button
            className="double-btn-item"
            aria-label="Создать"
            onClick={guestClick}
          >
            <PlusIcon size={20} />
          </button>
        </div>

        <main className="main-content">
          <section className="interests-section">
            <Tabs tabs={TABS} active={tab} onChange={setTab} />

            {postsLoading && posts.length === 0 ? (
              <div className="loading-feed">Загрузка...</div>
            ) : (
              <>
                <PublicationList
                  publications={posts}
                  currentUser={null}
                  onOpenPost={guestClick}
                  readOnly={true}
                  onRequireAuth={guestClick}
                />
                {hasMore && <div ref={sentinelRef} className="load-more-sentinel" />}
              </>
            )}

            <button type="button" className="landing-more" onClick={guestClick}>
              {authed ? 'Перейти в ленту' : 'Войдите, чтобы увидеть больше'}
            </button>

            <div className="landing-links">
              <Link to="/privacy.html" className="landing-link">Политика конфиденциальности</Link>
              <span className="landing-sep">·</span>
              <Link to="/terms.html" className="landing-link">Условия использования</Link>
            </div>
          </section>
        </main>
      </div>

      <SideMenu
        open={menuOpen}
        onClose={() => setMenuOpen(false)}
        items={NAV_ITEMS}
        onItemClick={handleNavClick}
      />
    </>
  );
}

  onClose,
  onOpenPost,
  initialEdit = false,
  forceSetup = false,
  onComplete,
}) {
  const [profile, setProfile] = useState(null);
  const [posts, setPosts] = useState([]);
  const [loading, setLoading] = useState(true);
  const [visible, setVisible] = useState(false);
  const [editing, setEditing] = useState(initialEdit);
  const [mode, setMode] = useState('normal'); /* 'normal' | 'full' */

  const scrollRef = useRef(null);
  const headerRef = useRef(null);
  const heroRef = useRef(null);
  const closingRef = useRef(false);
  const modeRef = useRef('normal');
  const panelRef = useRef(null);
  const panelDragRef = useRef({
    active: false, pointerId: null, startY: 0, lastY: 0, lastT: 0, velocity: 0, dy: 0,
  });
  const editSheetRef = useRef(null);
  const editClosingRef = useRef(false);
  const editDragRef = useRef({
    active: false, pointerId: null, startY: 0, lastY: 0, lastT: 0, velocity: 0, dy: 0,
  });

  /* ---------- управление переменными анимации ---------- */

  const setVar = useCallback((name, value) => {
    headerRef.current?.style.setProperty(name, value);
    heroRef.current?.style.setProperty(name, value);
  }, []);

  const setDragging = useCallback((on) => {
    [headerRef.current, heroRef.current].forEach((n) => {
      if (!n) return;
      if (on) n.setAttribute('data-dragging', '');
      else n.removeAttribute('data-dragging');
    });
  }, []);

  const applyMode = useCallback((next) => {
    modeRef.current = next;
    setMode(next);
    setVar('--t', next === 'full' ? '1' : '0');
  }, [setVar]);

  /* ---------- главное окно профиля: перетаскивание за ползунок ---------- */
  const onPanelDragStart = (e) => {
    if (forceSetup || closingRef.current) return;
    if (e.pointerType === 'mouse' && e.button !== 0) return;
    const d = panelDragRef.current;
    d.active = true;
    d.pointerId = e.pointerId;
    d.startY = e.clientY;
    d.lastY = e.clientY;
    d.lastT = performance.now();
    d.velocity = 0;
    d.dy = 0;
    if (panelRef.current) panelRef.current.style.transition = 'none';
    try { e.currentTarget.setPointerCapture(e.pointerId); } catch { /* ignore */ }
  };

  const onPanelDragMove = (e) => {
    const d = panelDragRef.current;
    if (!d.active || e.pointerId !== d.pointerId) return;
    const now = performance.now();
    const dt = now - d.lastT;
    if (dt > 0) d.velocity = (e.clientY - d.lastY) / dt;
    d.lastY = e.clientY;
    d.lastT = now;
    d.dy = Math.max(0, e.clientY - d.startY);
    if (panelRef.current) panelRef.current.style.transform = `translateY(${d.dy}px)`;
  };

  const onPanelDragEnd = (e) => {
    const d = panelDragRef.current;
    if (!d.active || e.pointerId !== d.pointerId) return;
    d.active = false;
    try { e.currentTarget.releasePointerCapture(e.pointerId); } catch { /* ignore */ }
    const el = panelRef.current;
    if (d.dy > 60 || d.velocity > 0.4) {
      handleClose();
    } else if (el) {
      el.style.transition = `transform 280ms ${SHEET_EASE}`;
      el.style.transform = 'translateY(0)';
      setTimeout(() => {
        if (closingRef.current || !panelRef.current) return;
        panelRef.current.style.transition = '';
        panelRef.current.style.transform = '';
      }, 290);
    }
  };

  /* ---------- шторка редактирования: закрытие и перетаскивание ---------- */
  const closeEdit = useCallback(() => {
    if (forceSetup || editClosingRef.current) return;
    const el = editSheetRef.current;
    if (!el) { setEditing(false); return; }
    editClosingRef.current = true;
    el.style.transition = `transform 320ms ${SHEET_EASE}`;
    el.style.transform = 'translateY(100%)';
    setTimeout(() => {
      setEditing(false);
      editClosingRef.current = false;
    }, 320);
  }, [forceSetup]);

  const onEditDragStart = (e) => {
    if (forceSetup || editClosingRef.current) return;
    if (e.pointerType === 'mouse' && e.button !== 0) return;
    const d = editDragRef.current;
    d.active = true;
    d.pointerId = e.pointerId;
    d.startY = e.clientY;
    d.lastY = e.clientY;
    d.lastT = performance.now();
    d.velocity = 0;
    d.dy = 0;
    if (editSheetRef.current) editSheetRef.current.style.transition = 'none';
    try { e.currentTarget.setPointerCapture(e.pointerId); } catch { /* ignore */ }
  };

  const onEditDragMove = (e) => {
    const d = editDragRef.current;
    if (!d.active || e.pointerId !== d.pointerId) return;
    const now = performance.now();
    const dt = now - d.lastT;
    if (dt > 0) d.velocity = (e.clientY - d.lastY) / dt;
    d.lastY = e.clientY;
    d.lastT = now;
    d.dy = Math.max(0, e.clientY - d.startY);
    if (editSheetRef.current) editSheetRef.current.style.transform = `translateY(${d.dy}px)`;
  };

  const onEditDragEnd = (e) => {
    const d = editDragRef.current;
    if (!d.active || e.pointerId !== d.pointerId) return;
    d.active = false;
    try { e.currentTarget.releasePointerCapture(e.pointerId); } catch { /* ignore */ }
    if (d.dy > 60 || d.velocity > 0.4) {
      closeEdit();
    } else if (editSheetRef.current) {
      editSheetRef.current.style.transition = `transform 280ms ${SHEET_EASE}`;
      editSheetRef.current.style.transform = 'translateY(0)';
    }
  };

  /* ---------- back button ---------- */
  useBackButton(() => {
    if (modeRef.current === 'full') { applyMode('normal'); return; }
    if (editing && !forceSetup) { closeEdit(); return; }
    if (!forceSetup) handleClose();
  }, true);

  /* ---------- slide up ---------- */
  useEffect(() => {
    requestAnimationFrame(() => requestAnimationFrame(() => setVisible(true)));
    const prev = document.body.style.overflow;
    document.body.style.overflow = 'hidden';
    return () => { document.body.style.overflow = prev; };
  }, []);

  /* ---------- раскрытие/сворачивание фото: свайп, колесо мыши, тачпад ---------- */
  useEffect(() => {
    const el = scrollRef.current;
    if (!el) return undefined;

    let startY = 0;
    let anchorY = null; /* точка, с которой начали тянуть, когда scrollTop стал 0 */
    let dragging = false;
    let lastT = 0;

    const begin = () => {
      if (dragging) return;
      dragging = true;
      setDragging(true);
    };

    const onStart = (e) => {
      if (e.touches.length !== 1) return;
      startY = e.touches[0].clientY;
      anchorY = el.scrollTop <= 0 ? startY : null;
      dragging = false;
    };

    const onMove = (e) => {
      if (e.touches.length !== 1) return;
      const y = e.touches[0].clientY;

      if (modeRef.current === 'normal') {
        if (el.scrollTop > 0) { anchorY = null; return; }
        if (anchorY === null) anchorY = y;
        const dy = y - anchorY;
        if (dy <= 0 && !dragging) return; /* обычный скролл вверх */
        if (e.cancelable) e.preventDefault();
        begin();
        lastT = clamp(dy / PULL_RANGE);
      } else {
        const dy = y - startY;
        if (dy >= 0 && !dragging) return;
        if (e.cancelable) e.preventDefault();
        begin();
        lastT = clamp(1 + dy / PULL_RANGE);
      }
      setVar('--t', String(lastT));
    };

    const onEnd = () => {
      if (!dragging) return;
      dragging = false;
      anchorY = null;
      setDragging(false);
      const goFull = modeRef.current === 'normal' ? lastT > 0.3 : lastT > 0.7;
      applyMode(goFull ? 'full' : 'normal');
    };

    /* колесо мыши / тачпад: вверх на самом верху — раскрыть, вниз — свернуть */
    let wheelLockUntil = 0;
    const onWheel = (e) => {
      if (e.ctrlKey || Math.abs(e.deltaY) < 2) return;
      const now = performance.now();

      if (now < wheelLockUntil) {
        /* инерция после смены состояния — гасим, чтобы не прокрутило ленту */
        if (modeRef.current === 'full' || el.scrollTop <= 0) e.preventDefault();
        return;
      }

      if (modeRef.current === 'normal') {
        if (el.scrollTop > 0 || e.deltaY > 0) return; /* обычный скролл */
        e.preventDefault();
        wheelLockUntil = now + 500;
        applyMode('full');
      } else {
        e.preventDefault();
        if (e.deltaY < 0) return;
        wheelLockUntil = now + 500;
        applyMode('normal');
      }
    };

    el.addEventListener('wheel', onWheel, { passive: false });
    el.addEventListener('touchstart', onStart, { passive: true });
    el.addEventListener('touchmove', onMove, { passive: false });
    el.addEventListener('touchend', onEnd);
    el.addEventListener('touchcancel', onEnd);
    return () => {
      el.removeEventListener('wheel', onWheel);
      el.removeEventListener('touchstart', onStart);
      el.removeEventListener('touchmove', onMove);
      el.removeEventListener('touchend', onEnd);
      el.removeEventListener('touchcancel', onEnd);
    };
  }, [applyMode, setVar, setDragging]);

  /* ---------- load profile + posts ---------- */
  const loadProfile = useCallback(() => {
    if (!user) return;
    supabase
      .from('profiles')
      .select('*')
      .eq('id', user.id)
      .maybeSingle()
      .then(({ data }) => setProfile(data));
  }, [user]);

  useEffect(() => {
    if (!user) return;
    loadProfile();

    supabase
      .from('repositories')
      .select('*')
      .eq('user_id', user.id)
      .order('created_at', { ascending: false })
      .then(({ data }) => {
        setPosts(data || []);
        setLoading(false);
      });
  }, [user, loadProfile]);

  const handleClose = () => {
    if (closingRef.current) return;
    closingRef.current = true;
    const el = panelRef.current;
    if (el) {
      el.style.transition = `transform 300ms ${SHEET_EASE}`;
      el.style.transform = 'translateY(100%)';
    }
    setVisible(false);
    setTimeout(() => onClose?.(), 300);
  };

  const handleEditComplete = () => {
    loadProfile();
    onComplete?.();
    if (forceSetup) setEditing(false);
    else closeEdit();
  };

  /* скролл: 1 → 0, без setState */
  const handleScroll = (e) => {
    setVar('--p', String(clamp(e.currentTarget.scrollTop / COLLAPSE_DISTANCE)));
  };

  const isVerified = user?.email && VERIFIED_EMAILS.includes(user.email);
  const displayName = profile?.display_name || profile?.username || user?.email?.split('@')[0] || 'user';
  const username = profile?.username || '';
  const avatarUrl = profile?.avatar_url;

  return createPortal(
    <div className={`profile-screen-root ${visible ? 'is-on' : ''}`}>
      <div
        className="profile-screen-backdrop"
        onClick={forceSetup ? undefined : handleClose}
      />

      <div
        ref={panelRef}
        className={`profile-screen-panel ${mode === 'full' ? 'is-full' : ''}`}
      >
        {/* ползунок: тянуть вниз, чтобы закрыть */}
        <div
          className={`profile-sheet-handle-zone ${forceSetup ? 'is-locked' : ''}`}
          onPointerDown={onPanelDragStart}
          onPointerMove={onPanelDragMove}
          onPointerUp={onPanelDragEnd}
          onPointerCancel={onPanelDragEnd}
        >
          <div className="profile-sheet-handle" />
        </div>

        <header className="profile-screen-header" ref={headerRef}>
          {!forceSetup && (
            <button className="profile-screen-btn" onClick={handleClose} aria-label="Назад">
              <BackIcon size={22} />
            </button>
          )}
          <div className="profile-screen-title">
            <Avatar url={avatarUrl} name={displayName} seed={user?.id} size={28} />
            <span className="profile-screen-title-text">{displayName}</span>
          </div>
          <button
            className="profile-screen-btn"
            onClick={async () => { await supabase.auth.signOut(); }}
            aria-label="Выйти"
          >
            <LogoutIcon size={20} />
          </button>
        </header>

        <div
          className="profile-screen-scroll"
          ref={scrollRef}
          onScroll={handleScroll}
        >
          <div className="profile-hero" ref={heroRef}>
            {/* состояние 2: фото на всю ширину */}
            <div className="profile-cover" aria-hidden="true">
              {avatarUrl ? (
                <img src={avatarUrl} alt="" className="profile-cover-img" draggable={false} />
              ) : (
                <div
                  className="profile-cover-fallback"
                  style={{ background: colorFromString(user?.id || displayName) }}
                >
                  <span>{initialFromString(displayName)}</span>
                </div>
              )}
              <div className="profile-cover-scrim" />
              <div className="profile-cover-info">
                <div className="profile-cover-name">
                  {displayName}
                  {isVerified && <VerifiedIcon size={20} />}
                </div>
                {username && <div className="profile-cover-username">@{username}</div>}
              </div>
            </div>

            {/* состояние 1 (и 0 при скролле): круглая аватарка */}
            <div className="profile-hero-normal">
              <div className="profile-hero-avatar">
                <Avatar url={avatarUrl} name={displayName} seed={user?.id} size={96} />
              </div>
              <div className="profile-hero-info">
                <div className="profile-hero-name">
                  {displayName}
                  {isVerified && <VerifiedIcon size={16} className="profile-hero-verified" />}
                </div>
                {username && (
                  <div className="profile-hero-username">@{username}</div>
                )}
              </div>
            </div>
          </div>

          <div className="profile-actions">
            <button className="profile-action" onClick={() => setEditing(true)}>
              Изменить
            </button>
          </div>

          <div className="profile-tabs">
            <button className="profile-tab is-active">Публикации</button>
          </div>

          {loading ? (
            <div className="profile-empty">
              <div className="profile-empty-sub">Загрузка…</div>
            </div>
          ) : posts.length === 0 ? (
            <div className="profile-empty">
              <div className="profile-empty-title">Публикаций пока нет…</div>
              <div className="profile-empty-sub">
                Опубликуйте фото или видео в своём профиле
              </div>
            </div>
          ) : (
            <div className="profile-posts">
              <PublicationList
                publications={posts}
                currentUser={user}
                onOpenPost={onOpenPost}
              />
            </div>
          )}
        </div>

        {/* editing bottom-sheet */}
        {editing && (
          <div className="profile-edit-sheet" ref={editSheetRef}>
            <div
              className={`profile-edit-handle-zone ${forceSetup ? 'is-locked' : ''}`}
              onPointerDown={onEditDragStart}
              onPointerMove={onEditDragMove}
              onPointerUp={onEditDragEnd}
              onPointerCancel={onEditDragEnd}
            >
              <div className="profile-edit-handle" />
            </div>
            <div className="profile-edit-sheet-content">
              <h3 className="profile-edit-title">
                {forceSetup ? 'Заполните профиль' : 'Редактировать профиль'}
              </h3>
              <ProfileSetup
                onComplete={handleEditComplete}
                submitLabel={forceSetup ? 'Готово' : 'Сохранить'}
              />
            </div>
          </div>
        )}
      </div>
    </div>,
    document.body
  );
}

export default function Lightbox({ file, onClose }) {
  const [visible, setVisible] = useState(false);
  const onCloseRef = useRef(onClose);
  onCloseRef.current = onClose;

  const close = () => {
    setVisible(false);
    setTimeout(() => onCloseRef.current?.(), 250);
  };

  /* android back button closes */
  useBackButton(close, true);

  useEffect(() => {
    requestAnimationFrame(() => requestAnimationFrame(() => setVisible(true)));
    const prev = document.body.style.overflow;
    document.body.style.overflow = 'hidden';

    const onKey = (e) => { if (e.key === 'Escape') close(); };
    window.addEventListener('keydown', onKey);
    return () => {
      document.body.style.overflow = prev;
      window.removeEventListener('keydown', onKey);
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  const isImage = file.type?.startsWith('image/');
  const isVideo = file.type?.startsWith('video/');
  const isAudio = file.type?.startsWith('audio/');

    <div className={`lightbox ${visible ? 'is-visible' : ''}`} onClick={close}>

const MAX_FILE_SIZE = 10 * 1024 * 1024;

const formatSize = (bytes) => {
  if (bytes === 0) return '0 B';
  const k = 1024;
  const i = Math.floor(Math.log(bytes) / Math.log(k));
  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + ['B', 'KB', 'MB', 'GB'][i];
};

const getIcon = (type) => {
  if (type.startsWith('image/')) return '🖼️';
  if (type.startsWith('video/')) return '🎬';
  if (type.startsWith('audio/')) return '🎵';
  return '📄';
};

/* compress an image via canvas, keeps original filename/extension */
const compressImage = (file, maxW = 1600, quality = 0.75) => {
  if (!file.type.startsWith('image/')) return Promise.resolve(file);
  if (file.type === 'image/gif' || file.type === 'image/svg+xml') return Promise.resolve(file);

  return new Promise((resolve) => {
    const img = new Image();
    const url = URL.createObjectURL(file);

    img.onload = () => {
      URL.revokeObjectURL(url);

      const canvas = document.createElement('canvas');
      let { width, height } = img;

      if (width > maxW) {
        height = Math.round((height * maxW) / width);
        width = maxW;
      }

      canvas.width = width;
      canvas.height = height;

      const ctx = canvas.getContext('2d');
      ctx.drawImage(img, 0, 0, width, height);

      const mime = file.type === 'image/png' ? 'image/png' : 'image/jpeg';

      canvas.toBlob(
        (blob) => {
          if (!blob || blob.size >= file.size) {
            resolve(file);
            return;
          }
          const out = new File([blob], file.name, { type: mime });
          resolve(out);
        },
        mime,
        quality
      );
    };

    img.onerror = () => {
      URL.revokeObjectURL(url);
      resolve(file);
    };

    img.src = url;
  });
};

        url: URL.createObjectURL(compressed),
      });

import React, { useState, useEffect, useRef } from 'react';

const formatNumber = (n) => {
  if (n >= 1e6) return (n / 1e6).toFixed(1) + 'M';
  if (n >= 1e3) return (n / 1e3).toFixed(1) + 'K';
  return n;
};

export default function PostFooter({ views, likes, isLiked, burst, onLike, files = [], postId, onOpenComments }) {
  const [slide] = useState(0);
  const [commentCount, setCommentCount] = useState(0);
  /* уникальный id инстанса — чтобы каналы не конфликтовали */
  const instanceIdRef = useRef(Math.random().toString(36).slice(2, 9));

  useEffect(() => {
    if (!postId) return;
    supabase
      .from('comments')
      .select('*', { count: 'exact', head: true })
      .eq('repository_id', postId)
      .then(({ count }) => setCommentCount(count || 0));
  }, [postId]);

  useEffect(() => {
    if (!postId) return;
    const uid = instanceIdRef.current;
    const ch = supabase
      .channel(`comment-count-${postId}-${uid}`)
      .on('postgres_changes',
        { event: '*', schema: 'public', table: 'comments', filter: `repository_id=eq.${postId}` },
        async () => {
          const { count } = await supabase
            .from('comments')
            .select('*', { count: 'exact', head: true })
            .eq('repository_id', postId);
          setCommentCount(count || 0);
        })
      .subscribe();
    return () => supabase.removeChannel(ch);
  }, [postId]);

  const handleDownload = async (e) => {
    e.stopPropagation();
    const f = files[slide] || files[0];
    if (!f) return;
    const filename = f.name || 'download';
    try {
      const res = await fetch(f.url);
      const blob = await res.blob();
      const a = document.createElement('a');
      const objUrl = URL.createObjectURL(blob);
      a.href = objUrl;
      a.download = filename;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      setTimeout(() => URL.revokeObjectURL(objUrl), 1000);
    } catch {
      window.open(f.url, '_blank');
    }
  };

  const handleComments = (e) => {
    e.stopPropagation();
    onOpenComments?.();
  };

  return (
    <div className="post-footer">
      <button
        className={`post-action like-btn ${isLiked ? 'is-liked' : ''} ${burst ? 'is-bursting' : ''}`}
        onClick={(e) => { e.stopPropagation(); onLike?.(); }}
        aria-label="Нравится"
      >
        <div className="like-icon-wrapper">
          {isLiked ? <HeartFilledIcon size={20} className="like-icon" /> : <HeartIcon size={20} className="like-icon" />}
        </div>
        <span>{formatNumber(likes)}</span>
      </button>

      <button className="post-action comment-btn" onClick={handleComments} aria-label="Комментарии">
        <CommentIcon size={20} />
        <span>{formatNumber(commentCount)}</span>
      </button>

      <div className="post-action post-views">
        <EyeIcon size={20} />
        <span>{formatNumber(views)}</span>
      </div>

      <button className="post-action post-download-btn" onClick={handleDownload} aria-label="Скачать">
        <DownloadIcon size={20} />
        <span>Скачать</span>
      </button>
    </div>
  );
}

import React, { useState, useEffect, useRef, useId } from 'react';
import { createPortal } from 'react-dom';
import { supabase } from '../../../../lib/supabaseClient';
import { getProfile } from '../../../../lib/profilesCache';
import PostHeader from './PostHeader';
import PostMedia from './PostMedia';
import PostFooter from './PostFooter';
import PostContextMenu from './PostContextMenu';
import Lightbox from '../media/Lightbox';
import './PostCard.css';

export default function PostCard({
  post,
  currentUser,
  onOpen,
  readOnly = false,
  onRequireAuth,
}) {
  const [views, setViews] = useState(post.views || 0);
  const [likes, setLikes] = useState(0);
  const [isLiked, setIsLiked] = useState(false);
  const [burst, setBurst] = useState(false);
  const [preview, setPreview] = useState(null);
  const [author, setAuthor] = useState(null);
  const [menuOpen, setMenuOpen] = useState(false);

  const uid = useId().replace(/:/g, '');
  const pendingLikeRef = useRef(false);
  const lastLocalRef = useRef(0);
  const mountedRef = useRef(true);
  const longPressRef = useRef({ timer: null, startX: 0, startY: 0 });

  const isAuthor = !!(currentUser && post.user_id === currentUser.id);
  const hasCaption = !!(post.name && post.name.trim());

  const requireAuth = () => {
    if (onRequireAuth) onRequireAuth();
  };

  useEffect(() => {
    mountedRef.current = true;
    return () => { mountedRef.current = false; };
  }, []);

  useEffect(() => {
    if (!post.user_id) return;
    let cancelled = false;
    getProfile(supabase, post.user_id).then((data) => {
      if (!cancelled && mountedRef.current && data) setAuthor(data);
    });
    return () => { cancelled = true; };
  }, [post.user_id]);

  useEffect(() => {
    if (readOnly) return;
    const load = async () => {
      if (pendingLikeRef.current) return;
      const { count } = await supabase
        .from('reactions')
        .select('*', { count: 'exact', head: true })
        .eq('repository_id', post.id);
      if (pendingLikeRef.current || !mountedRef.current) return;
      setLikes(count || 0);

      if (currentUser) {
        const { data } = await supabase
          .from('reactions')
          .select('id')
          .eq('repository_id', post.id)
          .eq('user_id', currentUser.id)
          .maybeSingle();
        if (!pendingLikeRef.current && mountedRef.current) setIsLiked(!!data);
      }
    };
    load();
  }, [post.id, currentUser, readOnly]);

  useEffect(() => {
    if (readOnly) return;
    const ch = supabase
      .channel(`post-${post.id}-${uid}`)
      .on('postgres_changes',
        { event: '*', schema: 'public', table: 'reactions', filter: `repository_id=eq.${post.id}` },
        async () => {
          if (pendingLikeRef.current || !mountedRef.current) return;
          if (Date.now() - lastLocalRef.current < 2000) return;
          const { count } = await supabase
            .from('reactions')
            .select('*', { count: 'exact', head: true })
            .eq('repository_id', post.id);
          if (mountedRef.current) setLikes(count || 0);
        })
      .on('postgres_changes',
        { event: 'UPDATE', schema: 'public', table: 'repositories', filter: `id=eq.${post.id}` },
        (p) => { if (mountedRef.current) setViews(p.new.views || 0); })
      .subscribe();
    return () => supabase.removeChannel(ch);
  }, [post.id, currentUser, uid, readOnly]);

  const handleLike = async () => {
    if (readOnly) { requireAuth(); return; }
    if (!currentUser || pendingLikeRef.current) return;
    pendingLikeRef.current = true;
    lastLocalRef.current = Date.now();

    const wasLiked = isLiked;
    const prevLikes = likes;
    setIsLiked(!wasLiked);
    setLikes(wasLiked ? prevLikes - 1 : prevLikes + 1);

    if (!wasLiked) {
      setBurst(true);
      setTimeout(() => setBurst(false), 600);
    }

    try {
      if (wasLiked) {
        await supabase.from('reactions').delete()
          .eq('repository_id', post.id).eq('user_id', currentUser.id);
      } else {
        await supabase.from('reactions').insert({
          repository_id: post.id,
          user_id: currentUser.id,
          emoji: '❤️',
        });
      }
    } catch {
      if (mountedRef.current) {
        setIsLiked(wasLiked);
        setLikes(prevLikes);
      }
    } finally {
      setTimeout(() => { pendingLikeRef.current = false; }, 500);
    }
  };

  const handleOpenComments = () => {
    if (readOnly) { requireAuth(); return; }
    onOpen?.(post);
  };

  /* skip click if it landed on an interactive element */
  const handleCardClick = (e) => {
    if (!readOnly) return;
    if (e.target.closest('button, a, input, textarea, [data-no-redirect]')) return;
    requireAuth();
  };

  const handleContextMenu = (e) => {
    if (readOnly) {
      e.preventDefault();
      requireAuth();
      return;
    }
    e.preventDefault();
    setMenuOpen(true);
  };

  const handlePointerDown = (e) => {
    if (e.pointerType === 'mouse') return;
    if (readOnly) return;
    longPressRef.current.startX = e.clientX;
    longPressRef.current.startY = e.clientY;
    longPressRef.current.timer = setTimeout(() => {
      longPressRef.current.timer = null;
      setMenuOpen(true);
    }, 500);
  };

  const handlePointerMove = (e) => {
    if (readOnly) return;
    const s = longPressRef.current;
    if (!s.timer) return;
    if (Math.abs(e.clientX - s.startX) > 10 || Math.abs(e.clientY - s.startY) > 10) {
      clearTimeout(s.timer);
      s.timer = null;
    }
  };

  const handlePointerEnd = () => {
    if (readOnly) return;
    const s = longPressRef.current;
    if (s.timer) {
      clearTimeout(s.timer);
      s.timer = null;
    }
  };

  const handleShare = async () => {
    const postUrl = `${window.location.origin}/post/${post.id}`;
    if (navigator.share) {
      try {
        await navigator.share({
          title: post.name || 'Пост в «связь»',
          text: post.name || '',
          url: postUrl,
        });
        return;
      } catch (e) {
        if (e.name === 'AbortError') return;
      }
    }
    try {
      await navigator.clipboard.writeText(postUrl);
    } catch {
      const ta = document.createElement('textarea');
      ta.value = postUrl;
      document.body.appendChild(ta);
      ta.select();
      document.execCommand('copy');
      document.body.removeChild(ta);
    }
  };

  const handleCopy = async () => {
    try {
      await navigator.clipboard.writeText(post.name || '');
    } catch {
      const ta = document.createElement('textarea');
      ta.value = post.name || '';
      document.body.appendChild(ta);
      ta.select();
      document.execCommand('copy');
      document.body.removeChild(ta);
    }
  };

  const handleDelete = async () => {
    if (!window.confirm('Удалить пост? Это действие необратимо.')) return;
    try {
      const marker = '/repopart-files/';
      const paths = (post.files || [])
        .map((f) => {
          const idx = (f.url || '').indexOf(marker);
          return idx >= 0 ? f.url.slice(idx + marker.length) : null;
        })
        .filter(Boolean);

      if (paths.length) {
        await supabase.storage.from('repopart-files').remove(paths);
      }

      const { error } = await supabase.from('repositories').delete().eq('id', post.id);
      if (error) throw error;
    } catch (e) {
      alert('Ошибка удаления: ' + e.message);
    }
  };

  const fallbackName = post.author_email ? post.author_email.split('@')[0] : 'user';
  const displayName = author?.username || fallbackName;

  return (
    <>
      <div
        className={`post-card ${readOnly ? 'is-preview' : ''}`}
        onClick={handleCardClick}
        onContextMenu={handleContextMenu}
        onPointerDown={handlePointerDown}
        onPointerMove={handlePointerMove}
        onPointerUp={handlePointerEnd}
        onPointerCancel={handlePointerEnd}
      >
        <PostHeader
          author={author}
          userId={post.user_id}
          displayName={displayName}
          createdAt={post.created_at}
          authorEmail={post.author_email}
        />

        {post.name && <div className="post-text">{post.name}</div>}

        {post.files?.length > 0 && (
          <PostMedia
            files={post.files}
            postName={post.name}
            onPreview={readOnly ? requireAuth : setPreview}
          />
        )}

        <PostFooter
          views={views}
          likes={likes}
          isLiked={isLiked}
          burst={burst}
          onLike={handleLike}
          files={post.files}
          postId={post.id}
          onOpenComments={handleOpenComments}
        />
      </div>

      {!readOnly && (
        <PostContextMenu
          open={menuOpen}
          onClose={() => setMenuOpen(false)}
          onShare={handleShare}
          onCopy={handleCopy}
          onDelete={handleDelete}
          isAuthor={isAuthor}
          hasCaption={hasCaption}
        />
      )}

      {preview && !readOnly && createPortal(
        <Lightbox file={preview} onClose={() => setPreview(null)} />,
        document.body
      )}
    </>
  );
}

export default function PostMedia({ files = [], postName, onPreview }) {
  const [slide, setSlide] = useState(0);
  const [dragging, setDragging] = useState(false);
  const carouselRef = useRef(null);

  const dragRef = useRef({
    active: false,
    startX: 0,
    startScroll: 0,
    moved: false,
    pointerId: null,
  });

  const goToSlide = (i) => {
    if (!carouselRef.current) return;
    carouselRef.current.scrollTo({
      left: i * carouselRef.current.offsetWidth,
      behavior: 'smooth',
    });
  };

  const handleClickCapture = (e) => {
    if (dragRef.current.moved) {
      e.stopPropagation();
      e.preventDefault();
      dragRef.current.moved = false;
    }
  };

  const handlePointerDown = (e) => {
    if (e.pointerType === 'mouse' && e.button !== 0) return;
    const el = carouselRef.current;
    if (!el) return;

    try { el.setPointerCapture(e.pointerId); } catch {}

    dragRef.current.active = true;
    dragRef.current.startX = e.clientX;
    dragRef.current.startScroll = el.scrollLeft;
    dragRef.current.moved = false;
    dragRef.current.pointerId = e.pointerId;
    setDragging(true);
  };

  const handlePointerMove = (e) => {
    if (!dragRef.current.active) return;
    if (e.pointerId !== dragRef.current.pointerId) return;
    const el = carouselRef.current;
    if (!el) return;

    const dx = e.clientX - dragRef.current.startX;
    if (Math.abs(dx) > 5) dragRef.current.moved = true;

    el.scrollLeft = dragRef.current.startScroll - dx;
  };

  const handlePointerUp = (e) => {
    if (!dragRef.current.active) return;
    if (e.pointerId !== dragRef.current.pointerId) return;

    const el = carouselRef.current;
    try { el?.releasePointerCapture(e.pointerId); } catch {}

    dragRef.current.active = false;
    setDragging(false);

    if (!el || !dragRef.current.moved) return;

    const w = el.offsetWidth;
    const targetIndex = Math.round(el.scrollLeft / w);
    el.scrollTo({ left: targetIndex * w, behavior: 'smooth' });
  };

  if (files.length === 0) return null;

  const multiple = files.length > 1;

  return (
    <>
      <div className="post-media-wrapper">
        {multiple ? (
          <div
            className={`post-media-carousel ${dragging ? 'is-dragging' : ''}`}
            ref={carouselRef}
            onPointerDown={handlePointerDown}
            onPointerMove={handlePointerMove}
            onPointerUp={handlePointerUp}
            onPointerCancel={handlePointerUp}
            onClickCapture={handleClickCapture}
            onScroll={() => {
              const el = carouselRef.current;
              if (el && !dragRef.current.active) {
                setSlide(Math.round(el.scrollLeft / el.offsetWidth));
              }
            }}
          >
            {files.map((f, i) => (
              <div key={f.id || i} className="post-media-slide">
                <MediaItem file={f} postName={postName} onPreview={onPreview} />
              </div>
            ))}
          </div>
        ) : (
          <div className="post-media-single">
            <MediaItem file={files[0]} postName={postName} onPreview={onPreview} />
          </div>
        )}
      </div>

      {multiple && (
        <div className="post-media-dots">
          {files.map((_, i) => (
            <span
              key={i}
              className={`post-media-dot ${i === slide ? 'is-active' : ''}`}
              onClick={() => goToSlide(i)}
            />
          ))}
        </div>
      )}
    </>
  );
}

export default function AuthScreen() {
  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState('');
  const [loading, setLoading] = useState(false);
  const [showPassword, setShowPassword] = useState(false);

  useEffect(() => {
    document.body.classList.add('auth-lock-scroll');
    return () => document.body.classList.remove('auth-lock-scroll');
  }, []);

  useEffect(() => {
    const saved = localStorage.getItem('svyaz-username');
    if (saved) setUsername(saved);
  }, []);

  /* login: resolve username → email → signInWithPassword */
  const handleLogin = async (e) => {
    e.preventDefault();
    setError('');
    setLoading(true);

    const raw = username.trim().toLowerCase();
    let email = raw;

    if (!raw.includes('@')) {
      /* lookup email by username */
      const { data } = await supabase
        .from('profiles')
        .select('email')
        .ilike('username', raw)
        .maybeSingle();

      if (!data?.email) {
        setError('Юзернейм не найден');
        setLoading(false);
        return;
      }
      email = data.email;
    }

    const { error } = await supabase.auth.signInWithPassword({ email, password });

    if (error) setError('Неверный юзернейм, почта или пароль');
    else localStorage.setItem('svyaz-username', raw);

    setLoading(false);
  };

  const handleGoogle = async () => {
    setError('');
    setLoading(true);

    const { error } = await supabase.auth.signInWithOAuth({
      provider: 'google',
      options: { redirectTo: window.location.origin + '/feed' },
    });

    if (error) {
      setError(error.message);
      setLoading(false);
    }
  };

  return (
    <div className="auth-overlay">
      <div className="auth-logo">
        <div className="auth-logo-cell" />
        <div className="auth-logo-glass">
          <Logo className="auth-logo-icon" />
          <span className="auth-logo-text">связь</span>
        </div>
        <div className="auth-logo-cell auth-logo-cell-right">
          <span className="auth-beta-capsule">BETA</span>
        </div>
      </div>

      <div className="auth-container">
        <LoginForm
          username={username}
          setUsername={setUsername}
          password={password}
          setPassword={setPassword}
          showPassword={showPassword}
          setShowPassword={setShowPassword}
          error={error}
          loading={loading}
          onSubmit={handleLogin}
          onGoogle={handleGoogle}
        />
      </div>
    </div>
  );
}

  if (loading) return <div className="loading-screen">Загрузка...</div>;

function AuthRoute({ user, loading }) {

import { supabase } from '../../lib/supabaseClient';
import Logo from '../../assets/icons/Logo';
import { LogoutIcon, PlusIcon } from '../../assets/icons/Icons';
import SideMenu from './components/layout/SideMenu';
import Tabs from './components/layout/Tabs';
import CreateRepoModal from './components/modals/CreateRepoModal';
import PublicationList from './components/post/PublicationList';
import PostDetailScreen from '../Post/PostDetailScreen';
import ProfileScreen from '../Profile/ProfileScreen';
import './FeedScreen.css';

const POSTS_PER_PAGE = 10;

const NAV_ITEMS = [
  { label: 'Лента', icon: '📡', action: 'feed' },
  { label: 'Профиль', icon: '👤', action: 'profile' },
  { label: 'Мои посты', icon: '📁', action: 'my' },
  { label: 'Настройки', icon: '⚙️', action: 'settings' },
];

const TABS = [
  { id: 'for-you', label: 'Для вас' },
  { id: 'fresh', label: 'Свежее' },
  { id: 'top', label: 'Топ' },
];

export default function FeedScreen() {
  const [posts, setPosts] = useState([]);
  const [loading, setLoading] = useState(false);
  const [refreshing, setRefreshing] = useState(false);
  const [hasMore, setHasMore] = useState(true);
  const [tab, setTab] = useState('for-you');
  const [menuOpen, setMenuOpen] = useState(false);
  const [modalOpen, setModalOpen] = useState(false);
  const [currentUser, setCurrentUser] = useState(null);
  const [openPost, setOpenPost] = useState(null);
  const [profileOpen, setProfileOpen] = useState(false);
  const [needsProfileSetup, setNeedsProfileSetup] = useState(false);

  const sentinelRef = useRef(null);
  const trackedIdsRef = useRef(new Set());
  const loadingRef = useRef(false);
  const requestIdRef = useRef(0);

  useEffect(() => {
    supabase.auth.getUser().then(({ data: { user } }) => setCurrentUser(user));
  }, []);

  useEffect(() => {
    if (!currentUser) return;
    supabase
      .from('profiles')
      .select('username')
      .eq('id', currentUser.id)
      .maybeSingle()
      .then(({ data }) => {
        if (!data?.username) setNeedsProfileSetup(true);
      });
  }, [currentUser]);

  const loadPosts = useCallback(async (mode, limit, offset) => {
    if (mode === 'for-you') {
      const { data, error } = await supabase.rpc('get_recommended_feed', {
        p_user_id: currentUser?.id || null,
        p_limit: limit,
        p_offset: offset,
      });
      if (error) return [];
      return data || [];
    }

    if (mode === 'top') {
      const { data } = await supabase
        .from('repositories')
        .select('*')
        .order('hot_score', { ascending: false })
        .order('created_at', { ascending: false })
        .range(offset, offset + limit - 1);
      return data || [];
    }

    const { data } = await supabase
      .from('repositories')
      .select('*')
      .order('created_at', { ascending: false })
      .range(offset, offset + limit - 1);
    return data || [];
  }, [currentUser]);

  const load = useCallback(async (reset = false) => {
    if (!reset && !hasMore) return;

    const reqId = ++requestIdRef.current;

    if (reset) {
      loadingRef.current = true;
      setPosts([]);
      setHasMore(true);
      setLoading(true);
    } else {
      if (loadingRef.current) return;
      loadingRef.current = true;
      setLoading(true);
    }

    const offset = reset ? 0 : posts.length;
    const data = await loadPosts(tab, POSTS_PER_PAGE, offset);

    if (reqId !== requestIdRef.current) return;

    if (reset) {
      setPosts(data);
      trackedIdsRef.current = new Set();
    } else {
      setPosts((prev) => [...prev, ...data]);
    }

    setHasMore(data.length === POSTS_PER_PAGE);
    setLoading(false);
    loadingRef.current = false;
  }, [tab, posts.length, hasMore, loadPosts]);

  useEffect(() => {
    load(true);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [tab, currentUser]);

  const handleLogoClick = useCallback(async () => {
    if (refreshing) return;

    const isAtTop = window.scrollY < 5;

    if (!isAtTop) {
      window.scrollTo({ top: 0, behavior: 'smooth' });
      return;
    }

    setRefreshing(true);
    const reqId = ++requestIdRef.current;

    try {
      const data = await loadPosts(tab, POSTS_PER_PAGE, 0);
      if (reqId !== requestIdRef.current) return;
      setPosts(data);
      trackedIdsRef.current = new Set();
      setHasMore(data.length === POSTS_PER_PAGE);
    } catch {
      // ignore
    }

    await new Promise((r) => setTimeout(r, 600));
    setRefreshing(false);
  }, [refreshing, tab, loadPosts]);

  useEffect(() => {
    if (!sentinelRef.current) return;
    if (!hasMore) return;

    const observer = new IntersectionObserver(
      (entries) => {
        if (entries[0].isIntersecting) load(false);
      },
      { rootMargin: '400px' }
    );
    observer.observe(sentinelRef.current);
    return () => observer.disconnect();
  }, [hasMore, load, tab]);

  useEffect(() => {
    const channel = supabase
      .channel('repositories-feed')
      .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'repositories' },
        (p) => setPosts((prev) => [p.new, ...prev]))
      .on('postgres_changes', { event: 'DELETE', schema: 'public', table: 'repositories' },
        (p) => setPosts((prev) => prev.filter((x) => x.id !== p.old.id)))
      .on('postgres_changes', { event: 'UPDATE', schema: 'public', table: 'repositories' },
        (p) => setPosts((prev) => prev.map((x) => x.id === p.new.id ? p.new : x)))
      .subscribe();
    return () => supabase.removeChannel(channel);
  }, []);

  useEffect(() => {
    if (!currentUser || posts.length === 0) return;
    const newIds = posts.map((p) => p.id).filter((id) => !trackedIdsRef.current.has(id));
    if (newIds.length === 0) return;
    newIds.forEach((id) => trackedIdsRef.current.add(id));
    supabase.rpc('track_views_batch', { p_post_ids: newIds, p_user_id: currentUser.id });
  }, [currentUser, posts.length]);

  const handleLogout = async () => {
    await supabase.auth.signOut();
  };

  const handleNavClick = (action) => {
    if (action === 'profile') setProfileOpen(true);
    setMenuOpen(false);
  };

  const handleOpenPost = (post) => {
    setOpenPost(post);
  };

  return (
    <>
      <div className="app">
        <div className="app-bg" />

        <aside className="desktop-sidebar">
          <ul className="side-menu-list">
            {NAV_ITEMS.map((item) => (
              <li key={item.label}>
                <button
                  className="side-menu-item"
                  onClick={() => handleNavClick(item.action)}
                >
                  <span className="side-menu-icon">{item.icon}</span>
                  {item.label}
                </button>
              </li>
            ))}
          </ul>
        </aside>

        <button className="menu-btn" aria-label="Меню" onClick={() => setMenuOpen(true)}>
          <div className="menu-btn-stripes">
            <span></span><span></span><span></span>
          </div>
        </button>

        <button className="fixed-logo" onClick={handleLogoClick} aria-label="Обновить ленту">
          <Logo className="logo-icon" />
          <span className="logo-text">связь</span>
          <span className="logo-beta-capsule">BETA</span>
        </button>

        <div className="double-btn">
          <button className="double-btn-item" aria-label="Выйти" onClick={handleLogout} title="Выйти">
            <LogoutIcon size={20} />
          </button>
          <button className="double-btn-item" aria-label="Создать" onClick={() => setModalOpen(true)}>
            <PlusIcon size={20} />
          </button>
        </div>

        <main className="main-content">
          <section className="interests-section">
            <Tabs tabs={TABS} active={tab} onChange={setTab} />

            {refreshing && (
              <div className="refresh-overlay">
                <div className="refresh-text">Обновляем рекомендации...</div>
              </div>
            )}

            {loading && posts.length === 0 ? (
              <div className="loading-feed">Загрузка...</div>
            ) : (
              <>
                <PublicationList
                  publications={posts}
                  currentUser={currentUser}
                  onOpenPost={handleOpenPost}
                />
                {hasMore && <div ref={sentinelRef} className="load-more-sentinel" />}
              </>
            )}
          </section>
        </main>
      </div>

      <SideMenu
        open={menuOpen}
        onClose={() => setMenuOpen(false)}
        items={NAV_ITEMS}
        onItemClick={handleNavClick}
      />

      {modalOpen && (
        <CreateRepoModal onClose={() => setModalOpen(false)} onCreate={() => load(true)} />
      )}

      {openPost && currentUser && (
        <PostDetailScreen
          post={openPost}
          currentUser={currentUser}
          onClose={() => setOpenPost(null)}
        />
      )}

      {needsProfileSetup && currentUser ? (
        <ProfileScreen
          user={currentUser}
          initialEdit={true}
          forceSetup={true}
          onClose={() => {}}
          onComplete={() => setNeedsProfileSetup(false)}
        />
      ) : profileOpen && currentUser ? (
        <ProfileScreen
          user={currentUser}
          onClose={() => setProfileOpen(false)}
          onOpenPost={handleOpenPost}
        />
      ) : null}
    </>
  );
}
